using CaptureReplica;
internal static class EditorChecksProgram
{
    [STAThread]
    private static int Main()
    {
        ApplicationConfiguration.Initialize();
        var results = EditorSelfTests.Run(Path.Combine(Path.GetTempPath(), "FSCaptureReplica-editor-checks"));
        foreach (var result in results) Console.WriteLine(result);
        return results.Any(x => x.StartsWith("FAIL")) ? 1 : 0;
    }
}
